%% Extrait du diagramme des exigences (req) du pont transbordeur.
%% Chaque exigence = une boîte à deux compartiments (stéréotype + nom / Id + Text).
%% Liens : décomposition (cercle barré d'une croix, côté exigence générale) et «refine».
\documentclass[tikz,border=4pt]{standalone}
\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,
  cadre/.style={draw=black!70, line width=0.9pt},
  lien/.style={line width=0.8pt, black!75},
  raffine/.style={-{Triangle[length=5pt,width=4.5pt]}, line width=0.8pt, black!75,
                  dash pattern=on 3pt off 2pt},
  boite/.style={draw=solideB, line width=0.9pt, fill=white, text width=2.3cm, inner sep=3pt,
                font=\scriptsize,
                execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
  tete/.style={boite, align=center},
  corps/.style={boite, align=left}]

%% ---- macro « exigence » : \exig{nom TikZ}{x}{y}{stéréotype}{nom}{Id}{Text} ----
\newcommand\exig[7]{%
  \node[tete, anchor=north] (#1h) at (#2,#3) {\textit{#4}\\[1pt] \textbf{#5}};
  \node[corps, anchor=north] (#1b) at (#1h.south) {Id = "#6"\\ Text = "#7"};}
%% ---- macro « cercle barré » (décomposition) --------------------------
\newcommand\decomp[2]{%
  \draw[lien, fill=white] (#1,#2) circle (0.13);
  \draw[lien] (#1-0.13,#2) -- (#1+0.13,#2) (#1,#2-0.13) -- (#1,#2+0.13);}

%% ================== cadre à onglet ====================================
\draw[cadre] (0,-0.25) rectangle (12.0,6.4);
\draw[cadre] (0,5.95) -- (2.35,5.95) -- (2.6,6.18) -- (2.6,6.4);
\node[font=\scriptsize, anchor=west, inner sep=3pt] at (0.1,6.17) {\texttt{req} [Paquet] Exigences};

%% ================== les cinq exigences ================================
\exig{r1}{1.45}{4.7}{«requirement»}{Franchissement de la Loire}{1}{Traverser la Loire à pied ou en busway}
\exig{r11}{4.35}{5.85}{«requirement»}{Transport de véhicules et piétons}{1.1}{Charge utile de la nacelle : 100 tonnes}
\exig{r111}{7.25}{5.85}{«functional Requirement»}{Traversée à pied}{1.1.1}{Deux cabines : 270 personnes}
\exig{r112}{7.25}{2.9}{«functional Requirement»}{Traversée en busway}{1.1.2}{Un busway par traversée}
\exig{r5}{10.6}{3.1}{«physical Requirement»}{Dimensions d'un busway}{5}{Largeur 2,55 m ; hauteur 3,12 m ; 150 personnes}

%% ================== liens de décomposition ============================
%% « 1 » se décompose en « 1.1 » : le cercle barré est du côté de l'exigence générale
\path let \p1 = (r1h.south east), \p2 = (r11h.south west) in
  \pgfextra{\xdef\ya{\y1}\xdef\yb{\y2}};
\decomp{2.9}{\ya}
\draw[lien] (2.9,\ya+0.13) -- (2.9,\yb) -- (r11h.south west);
%% « 1.1 » se décompose en « 1.1.1 »
\path let \p1 = (r11h.south east), \p2 = (r111h.south west) in
  \pgfextra{\xdef\yc{\y1}\xdef\yd{\y2}};
\decomp{5.75}{\yc}
\draw[lien] (5.75,\yc-0.13) -- (5.75,\yd) -- (r111h.south west);
%% « 1.1 » se décompose en « 1.1.2 »
\path let \p1 = (r11b.south), \p2 = (r112h.south west) in
  \pgfextra{\xdef\ye{\y1}\xdef\yf{\y2}};
\decomp{4.35}{\ye-0.2}
\draw[lien] (4.35,\ye) -- (4.35,\ye-0.07);
\draw[lien] (4.35,\ye-0.33) -- (4.35,\yf) -- (r112h.south west);

%% ================== lien «refine» =====================================
\draw[raffine] (9.35,2.3) -- (8.5,2.3);
\node[font=\scriptsize, anchor=south, inner sep=2pt] at (8.93,2.35) {«refine»};

%% ================== légende ===========================================
\decomp{0.35}{-0.6}
\node[font=\scriptsize, anchor=north west, align=left, inner sep=1pt] at (0.55,-0.45)
     {: l'exigence générale se décompose en exigences plus détaillées\\
      «refine» : précise l'exigence pointée};
\end{tikzpicture}
\end{document}
