\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,
  classe/.style={circle, minimum size=10mm, text=white, font=\small\bfseries, inner sep=0pt},
  lien/.style={line width=0.8pt}, etiquette/.style={font=\small, fill=white, inner sep=1.5pt}]
\node[classe, fill=black]   (E0) at (0,0)   {E0};
\node[classe, fill=solideE] (E1) at (0,3)   {E1};
\node[classe, fill=solideA] (E2) at (4.8,3) {E2};
\node[classe, fill=solideB] (E3) at (4.8,0) {E3};

\draw[lien] (E1) -- node[etiquette, above] {Pivot (A, $\vec{z}$)} (E2);
\draw[lien] (E2) -- node[etiquette, right] {Pivot (B, $\vec{z}$)} (E3);
\draw[lien] (E3) -- node[etiquette, below] {Pivot (D, $\vec{z}$)} (E0);
\end{tikzpicture}
