%% Projection d'un vecteur force sur les axes du repère.
%% F est dans le quadrant supérieur gauche : Fx < 0 et Fy > 0.
%% L'angle bêta est mesuré entre F et l'axe y.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,
  axe/.style={-{Stealth[length=5pt]}, line width=0.7pt, solideF},
  force/.style={-{Stealth[length=6pt]}, line width=1.4pt, solideA},
  compo/.style={-{Stealth[length=5pt]}, line width=1pt, solideA},
  rappel/.style={line width=0.5pt, dash pattern=on 3pt off 2pt, black!55}]

\coordinate (O) at (0,0);
\coordinate (F) at (-1.5,2.0);

% --- repère ------------------------------------------------------------
\draw[axe] (-2.15,0) -- (1.25,0) node[anchor=north, inner sep=3pt, font=\small, text=solideF] {$x$};
\draw[axe] (0,-0.5) -- (0,2.6) node[anchor=west, inner sep=3pt, font=\small, text=solideF] {$y$};
\node[font=\small, anchor=north west, inner sep=3pt] at (0.02,-0.02) {$O$};

% --- lignes de rappel et composantes -----------------------------------
\draw[rappel] (F) -- (-1.5,0);
\draw[rappel] (F) -- (0,2.0);
\draw[compo] (O) -- (-1.5,0) node[anchor=north, inner sep=4pt, font=\small, text=solideA] {$\vec{F_x}$};
\draw[compo] (O) -- (0,2.0) node[anchor=west, inner sep=4pt, font=\small, text=solideA] {$\vec{F_y}$};

% --- le vecteur force et l'angle bêta ----------------------------------
\draw[force] (O) -- (F) node[anchor=south east, inner sep=1pt, font=\small, text=solideA] {$\vec{F}$};
\draw[line width=0.6pt, black!65] (0,0.85) arc (90:127:0.85);
\node[font=\small, anchor=south, inner sep=1pt] at (-0.34,0.86) {$\beta$};

% --- base (i, j) -------------------------------------------------------
\draw[axe, line width=1.1pt] (0.75,-0.95) -- ++(0.5,0) node[anchor=north, inner sep=2pt, font=\small, text=solideF] {$\vec{\imath}$};
\draw[axe, line width=1.1pt] (0.75,-0.95) -- ++(0,0.5) node[anchor=west, inner sep=2pt, font=\small, text=solideF] {$\vec{\jmath}$};

% --- cartouche des projections -----------------------------------------
\node[draw=black!45, fill=black!3, rounded corners=3pt, inner sep=5pt, anchor=west,
      font=\small, align=left] at (1.55,1.35)
     {$F_x = -\,F \sin\beta$\\[2pt] $F_y = +\,F \cos\beta$\\[2pt] $F_z = 0$};
\end{tikzpicture}
\end{document}
