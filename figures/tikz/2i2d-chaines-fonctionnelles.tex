%% Les deux chaînes fonctionnelles d'un système et l'ACTION (Tle, p. 244-246).
%% Chaîne d'information (bleu) au-dessus, chaîne d'énergie (orange) en dessous ;
%% l'ordre issu du traitement arrive sur DISTRIBUER (et non sur ALIMENTER).
%% Mise en page « téléphone » : trois rangées de trois colonnes (la chaîne
%% d'énergie se poursuit à la ligne, comme un texte), ACTION en bas au centre.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,
  bloc/.style={line width=0.9pt, fill=white, rounded corners=2pt, minimum width=2cm,
               minimum height=0.62cm, inner sep=1pt, font=\scriptsize},
  bi/.style={bloc, draw=solideB},
  be/.style={bloc, draw=solideD},
  fi/.style={-{Stealth[length=5pt,width=4pt]}, line width=1pt, draw=solideB},
  fe/.style={-{Stealth[length=5pt,width=4pt]}, line width=1pt, draw=solideD},
  bandeau/.style={rounded corners=6pt, dash pattern=on 3pt off 2pt, line width=0.8pt},
  et/.style={font=\scriptsize, inner sep=2pt, align=center,
             execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}}]

%% ---- trois colonnes (x) et trois rangées (y) -------------------------
\def\colA{1.55} \def\colB{3.95} \def\colC{6.35}
\def\rgi{5.50}  % rangée « information »
\def\rga{4.00}  % rangée « énergie » 1
\def\rgb{2.45}  % rangée « énergie » 2

%% ---- flèche creuse « matière d'œuvre » (vers le bas) -----------------
%% #1 = x, #2 = ordonnée du talon, #3 = ordonnée de la pointe
\newcommand\moflec[3]{%
  \draw[draw=solideA, line width=0.9pt, fill=solideA!15]
    (#1-0.11,#2) -- (#1+0.11,#2) -- (#1+0.11,#3+0.26) -- (#1+0.22,#3+0.26)
    -- (#1,#3) -- (#1-0.22,#3+0.26) -- (#1-0.11,#3+0.26) -- cycle;}

%% ================== bandeau haut : chaîne d'information ===============
\draw[bandeau, draw=solideB!55] (0.45,5.10) rectangle (7.45,5.90);
\node[font=\scriptsize\bfseries, text=solideB, anchor=north west, inner sep=1pt]
     at (0.45,5.07) {Chaîne d'information};
\node[bi] (acq) at (\colA,\rgi) {ACQUÉRIR};
\node[bi] (tra) at (\colB,\rgi) {TRAITER};
\node[bi] (com) at (\colC,\rgi) {COMMUNIQUER};
\draw[fi] (acq) -- (tra);
\draw[fi] (tra) -- (com);

%% ================== bandeau bas : chaîne d'énergie (deux rangées) =====
\draw[bandeau, draw=solideD!55] (0.45,2.05) rectangle (7.45,4.45);
\node[font=\scriptsize\bfseries, text=solideD, anchor=north west, inner sep=1pt]
     at (0.45,2.02) {Chaîne d'énergie};
\node[be] (ali) at (\colA,\rga) {ALIMENTER};
\node[be] (dis) at (\colB,\rga) {DISTRIBUER};
\node[be] (con) at (\colC,\rga) {CONVERTIR};
\node[be] (tr2) at (\colA,\rgb) {TRANSMETTRE};
\node[be] (agi) at (\colB,\rgb) {AGIR};
\draw[fe] (ali) -- (dis);
\draw[fe] (dis) -- (con);
\draw[fe] (tr2) -- (agi);
%% passage à la ligne : CONVERTIR se poursuit sur TRANSMETTRE
\draw[fe, rounded corners=4pt] (con.south) -- (\colC,3.20) -- (\colA,3.20) -- (tr2.north);

%% ================== ordres : l'information commande DISTRIBUER ========
\draw[-{Stealth[length=7pt,width=7pt]}, line width=2.4pt, solideD]
     (\colB,5.14) -- (\colB,4.36);
\node[et, text=solideD, anchor=west, inner sep=1pt] at (4.08,4.76) {\textbf{Ordres}};

%% ================== entrée et sortie de la chaîne d'information =======
\draw[fi] (\colA,6.26) -- (acq.north);
\node[et, anchor=south, inner sep=1pt] at (\colA,6.29) {Consignes, commandes};
\draw[fi] (com.north) -- (\colC,6.26);
\node[et, anchor=south, inner sep=1pt] at (\colC,6.29) {Messages};

%% ================== zone ACTION et matière d'œuvre ====================
\draw[draw=solideA, line width=1.1pt, fill=solideA!8] (3.05,0.55) rectangle (5.65,1.35);
\node[font=\small\bfseries, text=solideA] at (4.35,0.95) {ACTION};
\draw[fe] (agi.south) -- (\colB,1.35);
\moflec{5.15}{1.95}{1.40}
\node[et, text=solideA, anchor=west, text width=2.4cm, align=left] at (5.34,1.66)
     {Matière d'œuvre entrante};
\moflec{5.15}{0.55}{0.00}
\node[et, text=solideA, anchor=west, text width=2.4cm, align=left] at (5.34,0.26)
     {Matière d'œuvre sortante};

%% ================== pertes ============================================
\draw[fe] (3.62,2.14) -- (2.92,1.66);
\node[et, text=solideD, anchor=north east, inner sep=1pt] at (2.88,1.63) {Pertes};

%% ================== retour de l'ACTION vers « Informations » ==========
\draw[fi, rounded corners=4pt] (3.05,0.95) -- (0.10,0.95) -- (0.10,\rgi) -- (acq.west);
\node[et, anchor=south west, inner sep=1pt] at (0.35,1.00) {Informations};
\end{tikzpicture}
\end{document}
