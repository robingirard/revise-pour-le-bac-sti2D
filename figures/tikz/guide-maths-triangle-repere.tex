%% Ferme de charpente : triangle ABC dans un repère orthonormé (unité : le mètre).
%% A(0;0), B(4;0), C(1;3). Les vecteurs AB (rouge) et AC (bleu) servent au produit scalaire ;
%% ni l'angle BAC ni la longueur BC ne sont donnés : ce sont les réponses attendues.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=1.3cm,y=1.3cm,
  axe/.style={-{Stealth[length=5pt]}, line width=0.6pt},
  vect/.style={-{Stealth[length=6pt]}, line width=1.3pt},
  grad/.style={font=\scriptsize}]

\coordinate (A) at (0,0);
\coordinate (B) at (4,0);
\coordinate (C) at (1,3);

%% ---------------- quadrillage et axes ---------------------------------
\draw[solideE!25, line width=0.3pt, step=1] (-0.5,-0.5) grid (5,3.5);
\draw[axe] (-0.55,0) -- (5.15,0) node[anchor=south east, inner sep=3pt, font=\small] {$x$};
\draw[axe] (0,-0.55) -- (0,3.65) node[anchor=south east, inner sep=2pt, font=\small] {$y$};

\foreach \i in {1,2,3,4} {\draw[line width=0.5pt] (\i,0) -- (\i,-0.12) node[grad, anchor=north, inner sep=2pt] {\i};}
\foreach \j in {1,2,3} {\draw[line width=0.5pt] (0,\j) -- (-0.12,\j) node[grad, anchor=east, inner sep=2pt] {\j};}
\node[font=\scriptsize, text=black!60, anchor=north west] at (0.95,-0.46) {unité : 1 m};

%% ---------------- le triangle ------------------------------------------
\filldraw[fill=solideA!8, draw=black, line width=0.7pt, line join=round]
   (A) -- (B) -- (C) -- cycle;

%% ---------------- vecteurs AB et AC ------------------------------------
\draw[vect, solideA] (A) -- (B);
\node[font=\small, text=solideA, anchor=north, inner sep=4pt] at (3.4,0) {$\overrightarrow{AB}$};
\draw[vect, solideB] (A) -- (C);
\node[font=\small, text=solideB, anchor=east, inner sep=3pt] at (0.45,1.7) {$\overrightarrow{AC}$};

%% ---------------- angle en A (sans valeur) ------------------------------
\draw[solideA, line width=0.9pt] (0.6,0) arc (0:71.57:0.6);
\node[font=\small, text=solideA, anchor=west, inner sep=2pt] at (36:0.72) {$\widehat{BAC}$};

%% ---------------- côté BC ----------------------------------------------
\draw[line width=1pt] (B) -- (C);
\node[font=\small, anchor=south west, inner sep=2pt] at (2.62,1.60) {$BC=?$};

%% ---------------- sommets -----------------------------------------------
\fill (A) circle (2pt); \node[font=\small, anchor=north east, inner sep=3pt] at (A) {$A(0\,;0)=O$};
\fill (B) circle (2pt); \node[font=\small, anchor=north west, inner sep=3pt] at (B) {$B(4\,;0)$};
\fill (C) circle (2pt); \node[font=\small, anchor=south, inner sep=4pt] at (C) {$C(1\,;3)$};
\end{tikzpicture}
\end{document}
