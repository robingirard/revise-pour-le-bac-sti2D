%% Le transformateur : constitution (noyau de fer doux + deux bobinages)
%% et symbole normalisé, avec la relation m = N2/N1 = U2/U1 = I1/I2.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,
  fil/.style={line width=0.8pt, line cap=round, line join=round},
  bob/.style={solideB, line width=1pt, line cap=round},
  fleche/.style={-{Stealth[length=5pt]}, line width=0.9pt, draw=solideA},
  etiq/.style={font=\scriptsize, align=center, inner sep=2pt},
  titre/.style={font=\footnotesize\bfseries}]

%% ================== partie gauche : constitution =========================
\begin{scope}[shift={(0,0)}]
  % noyau : couronne rectangulaire grise
  \fill[black!15, even odd rule] (0,0) rectangle (2.4,2.2) (0.65,0.55) rectangle (1.75,1.65);
  \begin{scope}[even odd rule]
    \path[clip] (0,0) rectangle (2.4,2.2) (0.65,0.55) rectangle (1.75,1.65);
    \foreach \lx in {0.1,0.25,...,2.35} { \draw[black!35, line width=0.3pt] (\lx,0) -- (\lx,2.2); }
  \end{scope}
  \draw[fil, draw=black!55] (0,0) rectangle (2.4,2.2);
  \draw[fil, draw=black!55] (0.65,0.55) rectangle (1.75,1.65);
  % bobinages primaire (branche gauche) et secondaire (branche droite)
  \foreach \y in {1.9,1.6,1.3,1.0,0.7} {
    \draw[bob] (0,\y) arc (90:270:0.13 and 0.15);
    \draw[bob] (0.65,\y) arc (90:-90:0.13 and 0.15);
    \draw[bob] (1.75,\y) arc (90:270:0.13 and 0.15);
    \draw[bob] (2.4,\y) arc (90:-90:0.13 and 0.15); }
  % annotations
  \node[etiq, anchor=south] at (1.2,2.45) {Noyau de fer doux};
  \node[etiq, text=solideB, anchor=north] at (0.25,-0.2) {Circuit\\primaire\\$N_1$ spires};
  \node[etiq, text=solideB, anchor=north] at (2.15,-0.2) {Circuit\\secondaire\\$N_2$ spires};
  \node[titre, anchor=south] at (1.2,2.9) {Constitution};
\end{scope}

%% ================== partie droite : symbole ==============================
\begin{scope}[shift={(4.9,0)}]
  % noyau : trait vertical double
  \draw[fil] (0,0.35) -- (0,1.85)  (0.14,0.35) -- (0.14,1.85);
  % enroulements : quatre demi-cercles de chaque côté
  \foreach \k in {0,1,2,3} {
    \draw[bob] (-0.06,{1.72-0.32*\k}) arc (90:270:0.16);
    \draw[bob] (0.2,{1.72-0.32*\k})  arc (90:-90:0.16); }
  % bornes du primaire
  \draw[fil] (-0.06,1.72) -- (-0.95,1.72)  (-0.06,0.44) -- (-0.95,0.44);
  \fill (-0.95,1.72) circle (1.2pt);  \fill (-0.95,0.44) circle (1.2pt);
  \draw[fleche] (-0.85,1.92) -- (-0.4,1.92);
  \node[etiq, text=solideA, anchor=south] at (-0.62,1.94) {$I_1$};
  \draw[fleche] (-1.2,0.5) -- (-1.2,1.66);
  \node[etiq, text=solideA, anchor=east] at (-1.25,1.08) {$U_1$};
  % bornes du secondaire
  \draw[fil] (0.2,1.72) -- (1.09,1.72)  (0.2,0.44) -- (1.09,0.44);
  \fill (1.09,1.72) circle (1.2pt);  \fill (1.09,0.44) circle (1.2pt);
  \draw[fleche] (0.75,1.92) -- (1.2,1.92);
  \node[etiq, text=solideA, anchor=south] at (0.97,1.94) {$I_2$};
  \draw[fleche] (1.34,0.5) -- (1.34,1.66);
  \node[etiq, text=solideA, anchor=west] at (1.39,1.08) {$U_2$};
  % points homologues
  \fill (-0.22,1.72) circle (1.1pt);  \fill (0.36,1.72) circle (1.1pt);
  \node[titre, anchor=south] at (0.07,2.9) {Symbole};
\end{scope}

%% ================== cadre, séparateur et relation ========================
\draw[black!30, line width=0.6pt] (3.35,-1.5) -- (3.35,3.15);
\draw[black!30, line width=0.6pt, rounded corners=3pt] (-0.55,-1.5) rectangle (6.6,3.4);
\node[anchor=north, font=\small, align=center] at (3.0,-1.7)
  {$m = \dfrac{N_2}{N_1} = \dfrac{U_2}{U_1} = \dfrac{I_1}{I_2}$\qquad
   \footnotesize $m > 1$ : élévateur --- $m < 1$ : abaisseur};
\end{tikzpicture}
\end{document}
