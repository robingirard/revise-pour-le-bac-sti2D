%% sous 230 V, I = 87 A (pertes ~ 15 kW) ; sous 20 kV, I = 1,0 A (pertes 2,0 W).

\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,

  etiq/.style={font=\scriptsize, inner sep=2pt}]

%% ---- générateur (production) --------------------------------------------

\node[etiq, align=center, anchor=west, text=black!75] at (0.5,1) {Production\\$U$};

%% ---- ligne : deux fils, chacun de résistance R --------------------------
\draw[fil] (0,2) -- (2.1,2)  (3.1,2) -- (5.3,2);
\draw[fil] (0,0) -- (2.1,0)  (3.1,0) -- (5.3,0);
\draw[fil, fill=white] (2.1,1.75) rectangle (3.1,2.25);

%% ---- consommateur --------------------------------------------------------
\draw[fil] (5.3,2) -- (5.3,1.7)  (5.3,0.3) -- (5.3,0);

  {$P_{\text{pertes}} = R \times I^{2}$};
