\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm, solide/.style={line width=1.4pt, line cap=round, line join=round}]

\foreach \y in {1.7,1.9,2.1,2.3} { \draw[line width=0.5pt] (2.35,\y) -- (2.20,\y-0.15) (2.85,\y) -- (3.00,\y+0.15); }
