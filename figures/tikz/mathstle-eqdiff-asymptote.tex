%% Solutions de (E) : y' = -4y + 80, c'est-à-dire f(t) = k e^{-4t} + 20.
%% Quel que soit k, toutes les solutions convergent vers la solution constante
%% y = -b/a = 20 (régime permanent), parce que a = -4 < 0 : e^{at} tend vers 0.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=4.0cm,y=0.045cm,
  axe/.style={-{Stealth[length=5pt]}, line width=0.6pt},
  courbe/.style={line width=1.3pt, line join=round},
  grad/.style={font=\scriptsize, inner sep=0pt}]

%% ---------------- axes et graduations -------------------------------------
\draw[axe] (-0.04,0) -- (1.5,0) node[anchor=north east, inner sep=3pt, font=\small] {$t$};
\draw[axe] (0,-6) -- (0,118) node[anchor=south east, inner sep=3pt, font=\small] {$y$};
\node[font=\small, anchor=north east, inner sep=2pt] at (-0.01,-1) {$O$};
\foreach \xv/\lab in {0.5/{0{,}5}, 1/{1}} {
  \draw[line width=0.5pt] (\xv,0) -- ++(0,-0.07cm);
  \node[grad, anchor=north, yshift=-3pt] at (\xv,0) {$\lab$};}
\foreach \yv in {20,50,100} {
  \draw[line width=0.5pt] (0,\yv) -- ++(-0.07cm,0);
  \node[grad, anchor=east, xshift=-3pt] at (0,\yv) {$\yv$};}

%% ---------------- deux autres solutions ------------------------------------
\draw[courbe, solideA] plot[domain=0:1.4, samples=80, smooth] (\x,{80*exp(-4*\x)+20});
\draw[courbe, solideB] plot[domain=0:1.4, samples=80, smooth] (\x,{-20*exp(-4*\x)+20});

%% ---------------- solution constante k = 0 ---------------------------------
\draw[courbe, solideE] (0,20) -- (1.2,20);
\draw[courbe, solideE, dash pattern=on 4pt off 3pt] (1.2,20) -- (1.45,20);
\node[font=\small, text=solideE, anchor=west, inner sep=3pt] at (1.45,20)
     {$y=-\dfrac{b}{a}=20$};
\node[font=\scriptsize, text=black!60, anchor=north] at (0.8,11)
     {solution constante = régime permanent};

%% ---------------- conditions initiales --------------------------------------
\fill (0,100) circle (2pt);
\node[font=\scriptsize, anchor=west, inner sep=4pt] at (0,100) {$f(0)=100$};
\fill (0,0) circle (2pt);
\node[font=\scriptsize, anchor=north west, inner sep=4pt] at (0,-1) {$f(0)=0$};

%% ---------------- encadré ----------------------------------------------------
\node[draw=black!45, line width=0.5pt, rounded corners=2pt, fill=white, inner sep=4pt,
      anchor=north east, align=left, font=\small] at (1.42,114)
     {$f(t)=k\,\mathrm{e}^{at}-\dfrac{b}{a}$\\[3pt]
      {\scriptsize si $a<0$ : $\displaystyle\lim_{t\to+\infty}f(t)=-\frac{b}{a}$}};
\end{tikzpicture}
\end{document}
