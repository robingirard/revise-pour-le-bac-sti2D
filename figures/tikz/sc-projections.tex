%% Les trois vues d'une équerre percée, en projection européenne, et sa perspective.
%% Le NOM de chaque vue est la réponse : il passe par \rappel.
\documentclass[tikz,border=4pt]{standalone}
\usepackage{liaisons}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm,
  trait/.style={line width=1.1pt, draw=solideB, line join=round},
  cache/.style={line width=0.7pt, draw=black!55, dash pattern=on 4pt off 3pt},
  axeT/.style={line width=0.5pt, draw=black!55, dash pattern=on 8pt off 3pt on 2pt off 3pt},
  corresp/.style={line width=0.35pt, draw=black!25, densely dotted},
  nom/.style={font=\small\bfseries, text=solideB!80!black, inner sep=3pt}]

%% ================= vue de face (en haut à gauche) =================
\draw[trait] (0,0) -- (4,0) -- (4,1) -- (1,1) -- (1,3) -- (0,3) -- cycle;
\draw[cache] (2.25,0) -- (2.25,1);
\draw[cache] (3.25,0) -- (3.25,1);
\node[nom, anchor=north] at (2,-0.15) {\rappel{Vue de face}};

%% ================= vue de gauche (à droite, projection européenne) =================
\begin{scope}[xshift=5.4cm]
\draw[trait] (0,0) rectangle (2,3);
\draw[cache] (0.5,0) -- (0.5,1);
\draw[cache] (1.5,0) -- (1.5,1);
\node[nom, anchor=north] at (1,-0.15) {\rappel{Vue de gauche}};
\end{scope}

%% ================= vue de dessus (en dessous) =================
\begin{scope}[yshift=-4.1cm]
\draw[trait] (0,0) rectangle (4,2);
\draw[cache] (1,0) -- (1,2);
\draw[trait] (2.75,1) circle (0.5);
\draw[axeT] (2.75,0.25) -- (2.75,1.75);
\draw[axeT] (2.0,1) -- (3.5,1);
\node[nom, anchor=north] at (2,-0.15) {\rappel{Vue de dessus}};
\end{scope}

%% ================= lignes de correspondance =================
\foreach \x in {1, 2.25, 3.25} { \draw[corresp] (\x,-0.45) -- (\x,-1.95); }
\draw[corresp] (4.15,0) -- (5.25,0);
\draw[corresp] (4.15,1) -- (5.25,1);
\draw[corresp] (4.15,3) -- (5.25,3);

%% ================= la pièce en perspective =================
\begin{scope}[xshift=5.4cm, yshift=-4.1cm]
\def\dx{0.75} \def\dy{0.55}
\fill[solideB!8] (0,0) -- (3,0) -- (3,0.75) -- (0.75,0.75) -- (0.75,2.25) -- (0,2.25) -- cycle;
\draw[trait, line width=0.9pt] (0,0) -- (3,0) -- (3,0.75) -- (0.75,0.75) -- (0.75,2.25) -- (0,2.25) -- cycle;
\draw[trait, line width=0.9pt] (0,2.25) -- (\dx,2.25+\dy) -- (0.75+\dx,2.25+\dy) -- (0.75,2.25);
\draw[trait, line width=0.9pt] (0.75+\dx,2.25+\dy) -- (0.75+\dx,0.75+\dy) -- (3+\dx,0.75+\dy) -- (3,0.75);
\draw[trait, line width=0.9pt] (3,0) -- (3+\dx,\dy) -- (3+\dx,0.75+\dy);
\draw[trait, line width=0.8pt] (2.25,1.03) ellipse (0.28 and 0.15);
\node[nom, anchor=north, text=black!60] at (1.9,-0.15) {perspective};
\end{scope}
\end{tikzpicture}
\end{document}
