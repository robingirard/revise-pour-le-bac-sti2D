  tuyau/.style={-{Stealth[length=6pt]}, line width=1.4pt, draw=black!65,

  fluxfin/.style={-{Stealth[length=5pt]}, line width=0.9pt, draw=solideD},
  cote/.style={{Stealth[length=4pt]}-{Stealth[length=4pt]}, line width=0.7pt, draw=black!75},
  sous/.style={font=\scriptsize, text=black!65, inner sep=1pt},
  amb/.style={font=\small, inner sep=1pt, anchor=west},
  don/.style={font=\scriptsize, inner sep=1pt, anchor=west}]

%% ================= les deux ambiances =====================================
\fill[solideB!8] (0.3,0.35) rectangle (3.6,4.25);
\fill[solideA!8] (4.0,0.35) rectangle (7.6,4.25);

\node[amb, text=solideA] at (4.15,4.02) {air ambiant : 20 $^\circ$C};

%% ================= la paroi isolante ======================================
\begin{scope}
  \path[clip] (3.6,0.35) rectangle (4.0,4.25);

\node[font=\small, rotate=90, inner sep=1pt] at (3.80,1.62) {$S$, $\lambda$};

%% ================= flacons de vaccin ======================================

\draw[fluxfin] (5.20,1.90) -- (2.90,1.90);

\node[font=\small\bfseries, text=solideD, anchor=south, inner sep=2pt] at (5.00,2.44) {$\Phi$};

%% ================= boucle du fluide frigorigène ===========================

\node[organe, draw=black]   (comp) at (3.80,3.45) {Compresseur};
\node[organe, draw=solideA] (cond) at (6.40,3.45) {Condenseur};

\node[sous, anchor=north] at (2.20,0.78) {liquide $\rightarrow$ gaz};

\draw[tuyau] (comp.east) -- (cond.west);
\draw[tuyau] (cond.south) -- (det.north);
